TITLE: Connection Lifecycle & Cleanup

CATEGORY: patterns

TAGS:
cleanup
connection
disconnect
lifecycle
event
maid

DESCRIPTION:
Connections created with RBXScriptSignal:Connect should be tracked
when the feature has a lifecycle or can be unloaded.

RULES:
- Track every long-lived connection.
- Disconnect connections when the feature is disabled.
- Do not create duplicate connections when toggling a feature.
- Prefer a central cleanup manager for modules with multiple connections.

GOOD_PATTERN:
local connections = {}

local function track(connection)
    table.insert(connections, connection)
    return connection
end

local function cleanup()
    for _, connection in ipairs(connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end

    table.clear(connections)
end

BAD_PATTERN:
RunService.Heartbeat:Connect(function()
    -- connection is never stored
end)

USE_WHEN:
- UI systems
- toggleable systems
- character lifecycle systems
- reloadable modules

local function FetchKnowledgeFile(path)
    if not KnowledgeConfig.Enabled then
        return nil, "Knowledge Engine disabled"
    end

    local url = BuildKnowledgeURL(path)

    local success, result = pcall(function()
        return game:HttpGet(url)
    end)

    if not success then
        DebugLog(
            "ERROR",
            "KNOWLEDGE_HTTP",
            "Failed to fetch: " .. tostring(path)
        )

        return nil, tostring(result)
    end

    if type(result) ~= "string" or result == "" then
        return nil, "Empty response"
    end

    if #result > KnowledgeConfig.MaxDocumentSize then
        DebugLog(
            "WARN",
            "KNOWLEDGE_HTTP",
            "Document too large: " .. tostring(path)
        )

        return nil, "Document exceeds size limit"
    end

    return result
end

local function KnowledgeCachePath(path)
    local safeName = tostring(path)
        :gsub("[\\/]", "_")
        :gsub("[^%w%._%-]", "_")

    return KnowledgeConfig.CacheFolder .. "/" .. safeName
end
PART 10 — Save Cache
local function SaveKnowledgeCache(path, content)
    if typeof(writefile) ~= "function" then
        return false
    end

    EnsureKnowledgeFolder()

    local cachePath = KnowledgeCachePath(path)

    local success = pcall(function()
        writefile(cachePath, content)
    end)

    if success then
        return true
    end

    return false
end
PART 11 — Load Cache
local function LoadKnowledgeCache(path)
    if typeof(isfile) ~= "function" or typeof(readfile) ~= "function" then
        return nil
    end

    local cachePath = KnowledgeCachePath(path)

    local exists = false

    pcall(function()
        exists = isfile(cachePath)
    end)

    if not exists then
        return nil
    end

    local success, content = pcall(function()
        return readfile(cachePath)
    end)

    if success and type(content) == "string" and content ~= "" then
        return content
    end

    return nil
end

local KnowledgeDocuments = {}

local function ParseKnowledgeManifest(text)
    local documents = {}

    for line in tostring(text):gmatch("[^\r\n]+") do
        line = line:gsub("^%s+", ""):gsub("%s+$", "")

        if line ~= "" and not line:match("^#") then
            local path, title, keywords =
                line:match("^([^|]+)|([^|]+)|(.+)$")

            if path and title and keywords then
                local tags = {}

                for keyword in keywords:gmatch("[^,]+") do
                    keyword = keyword:gsub("^%s+", "")
                    keyword = keyword:gsub("%s+$", "")

                    table.insert(tags, string.lower(keyword))
                end

                table.insert(documents, {
                    Path = path,
                    Title = title,
                    Keywords = tags,
                    Content = nil,
                    Loaded = false
                })
            end
        end
    end

    return documents
end
PART 14 — Load Manifest
local function LoadKnowledgeManifest(forceRefresh)
    local text, source = GetKnowledgeDocument(
        KnowledgeConfig.Manifest,
        forceRefresh
    )

    if not text then
        DebugLog(
            "ERROR",
            "KNOWLEDGE_MANIFEST",
            "Manifest load failed: " .. tostring(source)
        )

        return false
    end

    KnowledgeDocuments = ParseKnowledgeManifest(text)

    DebugLog(
        "INFO",
        "KNOWLEDGE_MANIFEST",
        string.format(
            "Loaded %d knowledge documents from %s",
            #KnowledgeDocuments,
            tostring(source)
        )
    )

    return true
end

local function Tokenize(text)
    local tokens = {}

    text = string.lower(tostring(text or ""))

    for token in text:gmatch("[%w_]+") do
        if #token >= 2 then
            tokens[token] = true
        end
    end

    return tokens
end
PART 17 — Scoring Engine

นี่คือหัวใจของระบบ

local function ScoreKnowledgeDocument(query, document)
    local queryTokens = Tokenize(query)

    local score = 0

    local title = string.lower(document.Title or "")
    local path = string.lower(document.Path or "")

    for token in pairs(queryTokens) do

        if title:find(token, 1, true) then
            score = score + 8
        end

        if path:find(token, 1, true) then
            score = score + 5
        end

        for _, keyword in ipairs(document.Keywords or {}) do
            if keyword == token then
                score = score + 10
            elseif keyword:find(token, 1, true) then
                score = score + 4
            end
        end
    end

    return score
end

PART 18 — Search Knowledge
local function SearchKnowledge(query)
    if #KnowledgeDocuments == 0 then
        LoadKnowledgeManifest(false)
    end

    local results = {}

    for _, document in ipairs(KnowledgeDocuments) do
        local score = ScoreKnowledgeDocument(query, document)

        if score >= KnowledgeConfig.MinScore then
            table.insert(results, {
                Document = document,
                Score = score
            })
        end
    end

    table.sort(results, function(a, b)
        return a.Score > b.Score
    end)

    while #results > KnowledgeConfig.SearchLimit do
        table.remove(results)
    end

    return results
end

local function EnsureKnowledgeContent(document)
    if document.Content and document.Content ~= "" then
        return document.Content
    end

    local content = GetKnowledgeDocument(
        document.Path,
        false
    )

    if type(content) == "string" then
        document.Content = content
        document.Loaded = true

        return content
    end

    return nil
end

local function BuildKnowledgeContext(query)
    local results = SearchKnowledge(query)

    if #results == 0 then
        return "-- No relevant GitHub knowledge found."
    end

    local output = {
        "-- [[ PAYOMBOYZ GITHUB KNOWLEDGE CONTEXT ]]",
        "-- Query: " .. tostring(query),
        "-- Results: " .. tostring(#results),
        ""
    }

    for index, result in ipairs(results) do
        local document = result.Document
        local content = EnsureKnowledgeContent(document)

        if content then
            table.insert(
                output,
                string.format(
                    "===== KNOWLEDGE #%d | SCORE %d | %s =====",
                    index,
                    result.Score,
                    document.Title
                )
            )

            table.insert(output, content)
            table.insert(output, "")
        end
    end

    return table.concat(output, "\n")
end

local function ExpandKnowledgeQuery(query)
    local tokens = Tokenize(query)
    local expanded = {}

    for token in pairs(tokens) do
        expanded[token] = true

        local synonyms = KnowledgeSynonyms[token]

        if synonyms then
            for _, synonym in ipairs(synonyms) do
                expanded[synonym] = true
            end
        end
    end

    return expanded
end
